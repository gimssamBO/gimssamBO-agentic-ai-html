문제 1 — 카테고리별 성장률 분석 + 차트

code/ch24_business.py를 복사해 code/ch24_ex_growth.py를 만드세요. load_facts를 확장해 각 카테고리의 첫 달 대비 마지막 달 성장률(%) 을 계산하고, matplotlib로 카테고리별 최신 달 매출 막대그래프를 reports/category_sales.png로 저장하세요.

  • 사용 파일: data/monthly_sales.csv
  • 기대 결과: 콘솔에 성장률 상위/하위 카테고리명이 출력되고 reports/category_sales.png가 생성된다.

# -*- coding: utf-8 -*-
# 실행: python code/ch24_ex_growth.py
import sys, pathlib
sys.path.append(str(pathlib.Path(__file__).resolve().parent))
from common import DATA
import pandas as pd
import matplotlib
matplotlib.use("Agg")                       # 화면 없는 환경 대응
import matplotlib.pyplot as plt
import koreanize_matplotlib                  # 한글 폰트(OS 무관) — import 한 줄로 끝

REPORTS = pathlib.Path(__file__).resolve().parent.parent / "reports"

def category_growth():
    """각 카테고리의 첫 달 대비 마지막 달 성장률(%)을 계산해 dict로 반환."""
    df = pd.read_csv(DATA / "monthly_sales.csv").sort_values("month").reset_index(drop=True)
    first, last = df.iloc[0], df.iloc[-1]
    cats = [c for c in df.columns if c not in ("month", "total")]
    growth = {}
    for c in cats:
        if first[c] > 0:                                  # 0으로 나누기 방지
            growth[c] = round((last[c] - first[c]) / first[c] * 100, 1)
    return last, cats, growth

if __name__ == "__main__":
    last, cats, growth = category_growth()
    ranked = sorted(growth.items(), key=lambda x: x[1], reverse=True)   # 성장률 내림차순
    print("[카테고리 성장률]")
    for c, g in ranked:
        print(f"  {c}: {g:+.1f}%")
    print(f"\n성장률 1위: {ranked[0][0]} ({ranked[0][1]:+.1f}%)")
    print(f"성장률 꼴찌: {ranked[-1][0]} ({ranked[-1][1]:+.1f}%)")

    # 최신 달 카테고리별 매출 막대그래프
    REPORTS.mkdir(exist_ok=True)
    vals = [last[c] / 10000 for c in cats]
    fig, ax = plt.subplots(figsize=(9, 5))
    ax.bar(cats, vals, color="#4C72B0")
    ax.set_title(f"{last['month']} 카테고리별 매출")
    ax.set_ylabel("매출(만원)")
    plt.xticks(rotation=30, ha="right")
    plt.tight_layout()
    path = REPORTS / "category_sales.png"
    plt.savefig(path, dpi=120); plt.close(fig)
    print("[차트 저장]", path)

**해설**: 핵심은 **성장률도 코드로 계산**(LLM에 안 맡김)한다는 24강 원칙입니다. 첫 달(`iloc[0]`)과 마지막 달(`iloc[-1]`)을 비교해 카테고리별 성장률을 구하고, `sorted`로 순위를 매깁니다. `if first[c] > 0` 으로 0 나누기를 방지하는 방어 코딩도 잊지 마세요(06번 정신). 차트는 07번과 같은 패턴 — `Agg` 백엔드·한글 폰트·`savefig`+`close`. `g:+.1f`의 `+`는 양수에 `+` 부호를 붙여 증감을 명확히 보여 줍니다.

문제 2 — 특정 달 지정 리포트

generate_monthly_report(month="2026-03")처럼 특정 달을 지정해 리포트를 생성하도록 인자를 추가하고, 존재하지 않는 달이면 친절한 예외 메시지를 출력하게 하세요.

  • 검증: 존재하는 달과 없는 달(예: "1999-01")을 각각 호출해, 정상 리포트와 예외 메시지가 구분되어 출력되는지 확인.

# -*- coding: utf-8 -*-
# 실행: python code/ch24_ex_month.py
import sys, pathlib
sys.path.append(str(pathlib.Path(__file__).resolve().parent))
from common import DATA
from ch24_business import write_report, save_report
import pandas as pd

def load_facts_for(month: str) -> dict:
    """지정한 달의 facts를 계산한다. 없는 달이면 ValueError를 던진다."""
    df = pd.read_csv(DATA / "monthly_sales.csv").sort_values("month").reset_index(drop=True)
    if month not in df["month"].values:                        # 존재 확인
        available = ", ".join(df["month"].astype(str))
        raise ValueError(f"'{month}' 데이터가 없습니다. 사용 가능한 달: {available}")
    idx = df.index[df["month"] == month][0]
    if idx == 0:
        raise ValueError(f"'{month}'은 첫 달이라 전월 대비를 계산할 수 없습니다.")
    cur, prev = df.iloc[idx], df.iloc[idx - 1]                 # 지정 달과 그 전월
    cats = [c for c in df.columns if c not in ("month", "total")]
    cur_cats = cur[cats].sort_values(ascending=False)
    growth = (cur["total"] - prev["total"]) / prev["total"] * 100
    return {
        "month": cur["month"], "prev_month": prev["month"],
        "total": int(cur["total"]), "prev_total": int(prev["total"]),
        "growth_pct": round(growth, 1),
        "top_category": cur_cats.index[0], "top_value": int(cur_cats.iloc[0]),
        "second_category": cur_cats.index[1], "second_value": int(cur_cats.iloc[1]),
        "by_category": {k: int(v) for k, v in cur_cats.items()},
    }

def generate_monthly_report(month: str = None) -> pathlib.Path:
    """특정 달 리포트 생성(month=None이면 최신 달). 없는 달이면 친절한 메시지."""
    try:
        if month is None:
            from ch24_business import load_facts
            facts = load_facts()
        else:
            facts = load_facts_for(month)
    except ValueError as e:
        print(f"[안내] {e}")                  # 친절한 예외 메시지
        return None
    body = write_report(facts)
    return save_report(facts, body)

if __name__ == "__main__":
    print("[1] 존재하는 달 (2026-03)")
    p = generate_monthly_report("2026-03")
    print("  결과:", "저장됨 →", p if p else "생성 실패")

    print("\n[2] 없는 달 (1999-01)")
    p = generate_monthly_report("1999-01")
    print("  결과:", "저장됨 →", p if p else "생성 실패(예상된 동작)")
**예상 출력** 
[1] 존재하는 달 (2026-03)
  결과: 저장됨 → /Users/.../reports/monthly_sales_2026-03.md

[2] 없는 달 (1999-01)
[안내] '1999-01' 데이터가 없습니다. 사용 가능한 달: 2026-01, 2026-02, 2026-03, 2026-04, 2026-05
  결과: 생성 실패(예상된 동작)
**해설**: 핵심은 **두 가지 예외 케이스**를 친절히 다루는 것입니다 — (1) 없는 달, (2) 첫 달(전월이 없어 증감 계산 불가). 없는 달이면 **사용 가능한 달 목록**을 함께 안내해(06번의 "진단 정보" 정신), 사용자가 바로 올바른 값을 알 수 있게 합니다. `df.index[df["month"] == month][0]`로 지정 달의 위치를 찾고, 그 전월(`idx-1`)과 비교하죠. 예외는 `ValueError`로 던지고 호출부에서 잡아 메시지만 출력 — 프로그램은 죽지 않습니다.
